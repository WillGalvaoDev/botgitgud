"""Independent v003 review, from product root; writes only temporary test data."""
import dataclasses
import itertools
import json
import math
import subprocess
import sys
import tempfile
import types
from pathlib import Path
from unittest.mock import patch

sys.path[:0] = ["tests/unit", "tests/fixtures"]
import test_m2_3_comparability_integration as delivery
from botgitgud.analysis.cohort import COHORT_TARGET_N
from botgitgud.analysis.cohort_match import quarantine_conflicting_duplicates, hygienic_candidates
from botgitgud.analysis.comparability_provenance import encode_comparability_provenance, decode_comparability_provenance
from botgitgud.analysis.measurement import damage_reference_id
from botgitgud.domain.models import DEFAULT_MATCHING_POLICY_VERSION

from botgitgud.report.coaching_answer import render_coaching_answer
from botgitgud.report.text import render_report

def old_module(name, path):
    source = subprocess.check_output(["git", "show", f"b6241df:{path}"]).decode("utf-8")
    module = types.ModuleType(name)
    sys.modules[name] = module
    exec(compile(source, path, "exec"), module.__dict__)
    return module

old_match = old_module("v003_review_old_match", "src/botgitgud/analysis/cohort_match.py")
old_metrics = old_module("v003_review_old_metrics", "src/botgitgud/analysis/metric_observations.py")
old_measurement = old_module("v003_review_old_measurement", "src/botgitgud/analysis/measurement.py")

with tempfile.TemporaryDirectory(prefix="m23-v003-independent-") as td:
    root = Path(td)
    target, refs, _ = delivery._mixed_population_fixture()
    result, deps = delivery._run_analysis_with_fake_references(root / "mixed", target, refs)
    try:
        before, _ = old_match.match_cohort(target, refs, min_n=COHORT_TARGET_N, matching_policy_version=DEFAULT_MATCHING_POLICY_VERSION)
        before_metrics = old_metrics.compare_metrics(target, before, deps.catalog)
        before_ledger = old_measurement.compare_damage(target, tuple(before))
        after = result.comparability
        comparison = result.dps_gap.comparison
        assert result.dps_gap.accounting_status == "AVAILABLE"
        assert comparison.total_delta_dps is not None
        assert math.isclose(math.fsum([*comparison.ability_delta_dps.values(), comparison.support_delta_dps, comparison.residual_dps]), comparison.total_delta_dps, abs_tol=1e-9)
        rows = {
            "ledger_input": [len(before), after.ledger.n],
            "ledger_accepted": [before_ledger.reference_n, comparison.reference_n],
            **{k: [len(v.reference_ids), len(result.dps_gap.metric_comparisons[k].reference_ids)] for k, v in before_metrics.items()},
        }
        print("BEFORE_AFTER " + json.dumps(rows, sort_keys=True))
        print("AFTER_EXCLUSIONS " + json.dumps(dict(after.metrics["gross_ability_dps:1"].descriptive.exclusion_counts), sort_keys=True))
        print("ACCOUNTING " + json.dumps({"total_delta_dps": comparison.total_delta_dps, "accepted_n": comparison.reference_n}))
        assert rows["ledger_input"] == [31, 30]
        assert rows["gross_ability_dps:1"] == [17, 16]
        assert rows["aura_uptime_fraction:1"] == [31, 34]
    finally:
        deps.store.close()

    # Revalidate R3 with case/space-normalized names, cross-server ID closure,
    # and a non-kill sharing the excluded ID (must be counted by hygiene).
    target, clean = delivery._r1_counterexample_target_and_clean_refs()
    a = delivery._make_log(character_name="Duplicate", class_name="Warlock", spec_name="Demonology", fight_id=501, uptime=None)
    b = dataclasses.replace(a, uptimes={1: 0.5})
    c = dataclasses.replace(b, percentile=99.0)
    cross_server = dataclasses.replace(c, build=dataclasses.replace(c.build, server="OtherRealm"))
    nonkill = dataclasses.replace(c, fight=dataclasses.replace(c.fight, kill=False))
    payloads = []
    for index, order in enumerate(itertools.permutations([a, b, c])):
        candidates = [*clean, *order, cross_server, nonkill]
        result, deps = delivery._run_analysis_with_fake_references(root / f"r3-{index}", target, candidates)
        try:
            p = result.comparability
            excluded = set(p.hygiene.conflicting_duplicate_ids)
            assert excluded == {"REPORT:501:Duplicate"}
            assert p.hygiene.excluded_conflicting_duplicates == 4
            assert p.hygiene.excluded_non_kill == 1
            assert p.hygiene.n_input == p.hygiene.n_output + p.hygiene.excluded_conflicting_duplicates + p.hygiene.excluded_self + p.hygiene.excluded_non_kill + p.hygiene.deduped_pull + p.hygiene.deduped_player
            assert not excluded.intersection((*p.eligibility.eligible_ids, *p.eligibility.ineligible_ids, *p.eligibility.indeterminate_ids, *p.ledger.member_ids))
            for v in p.metrics.values():
                assert not excluded.intersection((*v.descriptive.member_ids, *v.aspirational.member_ids))
            assert build_report_contract(result, setup=result.setup_analysis).confianca.comparability == p
            assert decode_comparability_provenance(result.manifest.comparability_provenance_json) == p
            assert decode_comparability_provenance(deps.store.query("SELECT comparability_provenance_json FROM runs").row(-1)[0]) == p
            payloads.append(encode_comparability_provenance(p))
        finally:
            deps.store.close()
    assert len(set(payloads)) == 1
    print("R3_EXTENDED six permutations PASS; four quarantined, one non-kill, no ID leakage; contract/store roundtrip PASS")

    # Same normalized observation, differing raw names -> both IDs must be closed.
    renamed = dataclasses.replace(b, build=dataclasses.replace(b.build, character_name=" duplicate "))
    for order in itertools.permutations([a, renamed]):
        output, report = quarantine_conflicting_duplicates(target, order)
        assert output == []
        assert set(report.conflicting_ids) == {damage_reference_id(a), damage_reference_id(renamed)}
    print("NORMALIZED_ID both divergent raw IDs removed PASS")

    # Isolated mutation probe: no source edits. The permanent test should reject
    # a Discord renderer fabricating a ledger comparison on the insufficient case.
    fabricated = "Comparacao da coorte: mediana 123456.7 DPS; gap 99999.9 DPS."
    with patch.object(delivery, "render_coaching_answer", return_value=fabricated) as renderer:
        try:
            delivery.test_run_analysis_ledger_insufficient_with_sufficient_metric(root / "discord-mutation")
        except AssertionError as exc:
            assert "non-computed comparison" in str(exc), str(exc)
            assert renderer.call_count == 1
        else:
            raise AssertionError("permanent test accepted fabricated Discord comparison")
    print("R2_DISCORD_MUTATION permanent integration test REJECTED fabricated ledger comparison: PASS")

    # Capture actual renderer inputs: absence checks for these CLI sections
    # exercise default arguments, because the integration test omits the fields.
    calls = []
    actual = delivery.render_report
    def capture(*args, **kwargs):
        calls.append({"positional_args": len(args), "keyword_args": sorted(kwargs)})
        return actual(*args, **kwargs)
    with patch.object(delivery, "render_report", side_effect=capture):
        delivery.test_run_analysis_ledger_insufficient_with_sufficient_metric(root / "cli-arguments")
    assert calls == [{"positional_args": 6, "keyword_args": ["confidence", "core_abilities", "external_dps_context", "proc_analysis", "setup"]}]
    print("CLI_RENDER_ARGUMENTS " + json.dumps(calls))

    # Actual production render, including the optional fields omitted by the test.
    refs = [delivery._make_log(character_name=f"Long{i}", class_name="Warlock", spec_name="Demonology", fight_id=i+2, duration_s=390) for i in range(16)]
    result, deps = delivery._run_analysis_with_fake_references(root / "actual-render", target, refs)
    try:
        contract = build_report_contract(result, setup=result.setup_analysis)
        cli = render_report(result.header, result.comparisons, result.manifest, result.performance, result.dps_gap, result.top_actions, setup=contract.setup, confidence=contract.confianca, core_abilities=contract.core_abilities, proc_analysis=contract.proc_analysis, external_dps_context=contract.external_dps_context)
        discord = render_coaching_answer(contract)
        assert "Compara\u00e7\u00e3o indispon\u00edvel" in cli
        assert "NO_REFERENCES" in cli
        assert discord == "A an\u00e1lise n\u00e3o encontrou uma prioridade de coaching sustentada pelos dados."
        assert result.dps_gap.metric_comparisons["gross_ability_dps:1"].finding is not None
        Path("docs/submilestones/M2.3/independent-final-render.json").write_text(json.dumps({"ledger": result.comparability.ledger.state, "metric_finding": True, "cli": cli, "discord": discord}, ensure_ascii=False, indent=2), encoding="utf-8")
        print("ACTUAL_RENDER full CLI and Discord safe on the current insufficient-ledger fixture PASS")
    finally:
        deps.store.close()

print("Independent v003 probes completed without editing production or permanent tests.")
