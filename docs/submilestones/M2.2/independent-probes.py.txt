"""Run from product workspace: .venv/Scripts/python.exe -B docs/submilestones/M2.2/independent-probes.py.txt"""
import dataclasses
import json
import runpy
import sys

sys.path.insert(0, "tests/fixtures")
fixtures = runpy.run_path("tests/unit/test_m2_2_metric_population.py")
make = fixtures["_make_log"]
populate = fixtures["_populate"]
target = fixtures["_target"]()

bad = make(character_name="Duplicate", class_name="Warrior")
good = make(character_name="Duplicate", class_name="Mage")
eligibility_results = []
for refs in [(bad, good), (good, bad)]:
    result = populate(target, refs)
    eligibility_results.append({
        "decisions": [r.decision.value for r in fixtures["evaluate_references"](target, refs).results],
        "members": result.descriptive.members,
        "excluded": dict(result.descriptive.excluded_reasons),
    })
assert eligibility_results[0]["members"] == ("REPORT:1:Duplicate",)
assert eligibility_results[1]["members"] == ()

missing = make(character_name="Duplicate", uptime=None)
observed = make(character_name="Duplicate", uptime=0.5)
availability_n = [
    populate(target, refs, metric="aura_uptime_fraction").descriptive.n
    for refs in [(missing, observed), (observed, missing)]
]
assert availability_n == [0, 1]

refs = (
    make(character_name="Z", external_buffs=frozenset({10060})),
    make(character_name="A", duration_s=800),
)
outputs = [populate(target, order) for order in (refs, refs[::-1])]
key_orders = [list(p.descriptive.excluded_reasons) for p in outputs]
assert key_orders[0] != key_orders[1]
assert json.dumps(dataclasses.asdict(outputs[0])) != json.dumps(dataclasses.asdict(outputs[1]))

print(json.dumps({
    "duplicate_eligibility": eligibility_results,
    "duplicate_availability_n": availability_n,
    "excluded_key_orders": key_orders,
}, indent=2))
