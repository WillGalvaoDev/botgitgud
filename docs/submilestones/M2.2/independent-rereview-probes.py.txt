"""Independent re-review probes; run with Python -B from product workspace."""
import dataclasses
import json
import runpy
import sys

sys.path.insert(0, "tests/fixtures")
f = runpy.run_path("tests/unit/test_m2_2_metric_population.py")
from botgitgud.analysis.metric_observations import UNITS

make, target, refs = f["_make_log"], f["_target"], f["_refs"]
populate = f["_populate"]
evaluate = f["evaluate_references"]
rid = f["damage_reference_id"]
checks = 0
for pair, metric in [
    ((make(character_name="Duplicate", class_name="Warrior"), make(character_name="Duplicate")), "gross_ability_dps"),
    ((make(character_name="Duplicate", uptime=None), make(character_name="Duplicate", uptime=0.5)), "aura_uptime_fraction"),
]:
    t = target()
    for order in (pair, pair[::-1]):
        for function in (f["select_metric_population"], f["covariate_sensitivity"]):
            try:
                function(t, order, metric=metric, spell_id=1, eligibility=evaluate(t, order), catalog=None)
            except ValueError as exc:
                assert "collision with disagreeing data" in str(exc)
            else:
                raise AssertionError("ambiguous identity accepted")
            checks += 1

for metric in UNITS:
    for n in (0, 8, 16):
        t = target()
        duplicate = make(character_name="Same")
        duplicate_copy = dataclasses.replace(duplicate)
        base = refs(n)
        assert populate(t, (*base, duplicate, duplicate_copy), metric=metric) == populate(t, (*base, duplicate), metric=metric)
        mixed = (*base, make(character_name="Z", external_buffs=frozenset({10060})), make(character_name="A", duration_s=1000), make(character_name="M", dps=None))
        before = repr(mixed)
        forward = populate(t, mixed, metric=metric)
        reverse = populate(t, mixed[::-1], metric=metric)
        assert json.dumps(dataclasses.asdict(forward)) == json.dumps(dataclasses.asdict(reverse))
        for population in (forward.descriptive, forward.aspirational):
            assert list(population.excluded_reasons) == sorted(population.excluded_reasons)
            assert population.n == len(population.members)
        assert set(forward.aspirational.members) <= set(forward.descriptive.members)
        assert repr(mixed) == before
        checks += 1
    for field, value in (("duration_s", 1000), ("external_buffs", frozenset({10060})), ("item_level", 1), ("tier_pieces", 0)):
        t = target()
        base = refs(8)
        candidate = make(character_name="Candidate", **{field: value})
        actual = populate(t, (*base, candidate), metric=metric).descriptive
        expected = {rid(log) for log in base}
        if field in ("item_level", "tier_pieces") and metric in ("player_casts_per_minute", "aura_uptime_fraction"):
            expected.add(rid(candidate))
        assert set(actual.members) == expected
        assert not actual.relaxation_steps
        checks += 1
print(json.dumps({"independent_checks_passed": checks, "R1": "fixed", "R2": "fixed"}))
